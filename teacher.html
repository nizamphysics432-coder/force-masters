function updateChapterDropdown() { const select = document.getElementById("chapterFilter"); const currentValue = select.value; // Fixed list of your quiz chapters const chapters = [ { value: "all", label: "📚 All Chapters" }, { value: "Exploring Magnets", label: "🧲 Exploring Magnets — Grade 6" }, { value: "Adolescence: A Stage of Growth and Change", label: "🌱 Adolescence: A Stage of Growth and Change — Grade 7" } ]; select.innerHTML = ""; chapters.forEach(function(chapter) { const option = document.createElement("option"); option.value = chapter.value; option.textContent = chapter.label; select.appendChild(option); }); // Keep the previous selection if possible const exists = Array.from(select.options) .some(function(option) { return option.value === currentValue; }); if (exists) { select.value = currentValue; } else { select.value = "all"; } }
